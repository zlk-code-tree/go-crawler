# 组织架构树状结构佣金统计方案

## 一、需求说明

通过组织架构树状节点作为搜索条件，查询每个节点下的佣金统计数据。

选择一个节点后，返回：
- 该节点的汇总统计（包含所有下属节点的数据）
- 直接子节点的各自统计数据

```
查询"济宁"节点示例：
├── 汇总：订单1000单，佣金17500元（济宁所有数据）
├── 投放一组：订单600单，佣金10500元
├── 投放二组：订单300单，佣金5000元
└── 未分配：订单100单，佣金2000元
```

## 二、API接口设计

### 接口地址

```
GET /api/org/stats
```

### 请求参数

| 参数 | 类型 | 必填 | 说明 |
|------|------|------|------|
| orgId | int64 | 是 | 组织节点ID |
| preset | string | 否 | 预设时间，默认thisMonth |
| startDate | string | 否 | 自定义开始日期（优先级高于preset）|
| endDate | string | 否 | 自定义结束日期（优先级高于preset）|

**preset可选值**：today / yesterday / last7 / last30 / thisMonth / all

## 三、数据结构说明

### 佣金计算公式

```
佣金 = 销售额 - 留底 - 货款

其中：
- 销售额(pay_amount) = product_price × product_num
- 留底(reserve_amount) = product_price × product_num × 5%
- 货款(cost_amount) = base_price × product_num（从goods表获取）
```

### 数据关联关系

```
org_structure (组织架构)
    │
    └── id ← shop_org_relation.org_id
                    │
                    └── shop_id → shop.id
                                    │
                                    └── uid → order_info.talent_id
```

### 有效订单条件

```sql
AND after_sale_status NOT IN ('退款成功', '已全额退款', '待退货')
AND order_status != '已关闭'
AND main_order_no IS NOT NULL
AND main_order_no != ''
```

```go
// GetOrgStats 获取组织节点佣金统计（单节点+直接子节点）
func (s *OrgStructureService) GetOrgStats(ctx context.Context, orgId int64, startDate, endDate string) (*model.OrgStatsNode, error) {
	// 1. 获取当前节点信息
	node, err := s.GetById(ctx, orgId)
	if err != nil {
		return nil, fmt.Errorf("查询组织节点失败: %v", err)
	}
	if node == nil || node.Id == 0 {
		return nil, fmt.Errorf("组织节点不存在")
	}

	// 2. 获取该节点下所有店铺ID（递归获取所有后代节点的店铺）
	shopIds, err := s.GetShopsByOrgId(ctx, orgId)
	if err != nil {
		return nil, fmt.Errorf("获取店铺列表失败: %v", err)
	}

	// 3. 计算当前节点的汇总统计
	summary, err := s.calcShopStats(ctx, shopIds, startDate, endDate)
	if err != nil {
		return nil, fmt.Errorf("计算统计失败: %v", err)
	}

	// 4. 获取直接子节点
	childrenNodes, err := s.GetChildren(ctx, orgId)
	if err != nil {
		return nil, fmt.Errorf("获取子节点失败: %v", err)
	}

	// 5. 计算每个直接子节点的统计
	childStats := make([]model.OrgStatsNode, 0, len(childrenNodes))
	for _, child := range childrenNodes {
		childShopIds, err := s.GetShopsByOrgId(ctx, child.Id)
		if err != nil {
			continue
		}
		childSummary, err := s.calcShopStats(ctx, childShopIds, startDate, endDate)
		if err != nil {
			continue
		}
		childStats = append(childStats, model.OrgStatsNode{
			OrgId:   child.Id,
			OrgName: child.Name,
			OrgType: child.Type,
			Summary: *childSummary,
		})
	}

	return &model.OrgStatsNode{
		OrgId:    node.Id,
		OrgName:  node.Name,
		OrgType:  node.Type,
		Summary:  *summary,
		Children: childStats,
	}, nil
}

// calcShopStats 计算指定店铺列表的佣金统计
func (s *OrgStructureService) calcShopStats(ctx context.Context, shopIds []int, startDate, endDate string) (*model.StatsSummary, error) {
	if len(shopIds) == 0 {
		return &model.StatsSummary{}, nil
	}

	sql := `
		SELECT
			COUNT(DISTINCT o.main_order_no) AS order_count,
			COALESCE(SUM(o.pay_amount), 0) AS pay_amount,
			COALESCE(SUM(o.cost_amount), 0) AS cost_amount,
			COALESCE(SUM(o.reserve_amount), 0) AS reserve_amount
		FROM (
			SELECT
				o1.main_order_no,
				SUM(COALESCE(o1.product_price, 0) * COALESCE(o1.product_num, 0)) AS pay_amount,
				SUM(COALESCE(g.base_price, 0) * COALESCE(o1.product_num, 0)) AS cost_amount,
				SUM(COALESCE(o1.product_price, 0) * COALESCE(o1.product_num, 0) * 0.05) AS reserve_amount
			FROM order_info o1
			LEFT JOIN goods g ON o1.seller_code = g.goods_code AND g.status = 1
			LEFT JOIN shop s ON o1.talent_id = s.uid
			WHERE s.id IN (?)
			  AND o1.pay_finish_time >= ?
			  AND o1.pay_finish_time < ?
			  AND o1.main_order_no IS NOT NULL
			  AND o1.main_order_no != ''
			  AND o1.after_sale_status NOT IN ('退款成功', '已全额退款', '待退货')
			  AND o1.order_status != '已关闭'
			GROUP BY o1.main_order_no
		) o
	`

	result, err := g.DB().GetOne(ctx, sql, shopIds, startDate, endDate)
	if err != nil {
		return nil, fmt.Errorf("查询统计失败: %v", err)
	}

	payAmount := result["pay_amount"].Float64()
	costAmount := result["cost_amount"].Float64()
	reserveAmount := result["reserve_amount"].Float64()

	return &model.StatsSummary{
		OrderCount:    result["order_count"].Int(),
		PayAmount:     roundFloat2(payAmount, 2),
		CostAmount:    roundFloat2(costAmount, 2),
		ReserveAmount: roundFloat2(reserveAmount, 2),
		Commission:    roundFloat2(payAmount-reserveAmount-costAmount, 2),
	}, nil
}

// roundFloat2 四舍五入保留指定小数位
func roundFloat2(val float64, precision int) float64 {
	ratio := 1.0
	for i := 0; i < precision; i++ {
		ratio *= 10
	}
	return float64(int(val*ratio+0.5)) / ratio
}
```

```go
// Stats 组织节点佣金统计 GET /api/org/stats
func (c *OrgStructureController) Stats(ctx context.Context, req *v1.OrgStatsReq) (res *v1.OrgStatsRes, err error) {
	orgService := service.NewOrgStructureService()

	// 解析时间范围（复用OrderAnalysisService的方法）
	orderService := service.NewOrderAnalysisService()
	startDate, endDate := orderService.ParseTimeRange(req.Preset, req.StartDate, req.EndDate)

	// 获取统计
	statsNode, err := orgService.GetOrgStats(ctx, req.OrgId, startDate, endDate)
	if err != nil {
		return nil, gerror.New("查询失败: " + err.Error())
	}

	res = &v1.OrgStatsRes{
		OrgId:     statsNode.OrgId,
		OrgName:   statsNode.OrgName,
		OrgType:   statsNode.OrgType,
		StartDate: startDate,
		EndDate:   endDate,
		Summary:   statsNode.Summary,
		Children:  statsNode.Children,
	}
	return
}
```
