# 操作日志前端对接说明

## 一、API 接口说明

### 1. 获取操作日志列表
**接口地址：** `GET /api/system/log/operation`

**请求参数：**
```typescript
{
  page: number;      // 页码，默认 1
  pageSize: number;  // 每页条数，默认 20
  keyword?: string;  // 搜索关键词（可选）
}
```

**响应数据：**
```typescript
{
  code: 0,
  msg: "ok",
  total: 100,
  data: [
    {
      id: 1,
      user_id: 1,
      username: "admin",
      real_name: "张立昆",           // 新增：用户真实姓名
      module: "用户管理",
      action: "查询",
      method: "GET",
      url: "/api/system/user/list",
      params: "page=1&pageSize=20",
      ip: "192.168.1.100",
      location: "内网IP",            // 新增：操作地点
      user_agent: "Mozilla/5.0...",
      status: 1,                     // 1-成功，0-失败
      error_msg: "",
      duration: 125,                 // 执行时长（毫秒）
      permission_code: "system:user:list",
      created_at: "2026-04-18 10:30:00"
    }
  ]
}
```

### 2. 获取操作日志详情
**接口地址：** `GET /api/system/log/operation/detail`

**请求参数：**
```typescript
{
  id: number;  // 日志ID
}
```

**响应数据：**
```typescript
{
  code: 0,
  msg: "ok",
  data: {
    id: 1,
    user_id: 1,
    username: "admin",
    real_name: "张立昆",
    email: "admin@example.com",    // 新增：用户邮箱
    phone: "13800138000",          // 新增：用户手机号
    module: "用户管理",
    action: "查询",
    method: "GET",
    url: "/api/system/user/list",
    params: "page=1&pageSize=20",
    ip: "192.168.1.100",
    location: "内网IP",
    user_agent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",

    error_msg: "",
```

## 二、前端页面修改

### 1. API 接口定义（TypeScript）

创建或修改 `src/api/system/log.ts`：

```typescript
import { requestClient } from '#/api/request';

export interface OperationLog {
  id: number;
  user_id: number;
  username: string;
  real_name: string;        // 新增
  module: string;
  action: string;
  method: string;
  url: string;
  params: string;
  ip: string;
  location: string;         // 新增
  user_agent: string;
  status: number;
  error_msg: string;
  duration: number;
  permission_code: string;
  created_at: string;
}

export interface OperationLogDetail extends OperationLog {
  email: string;            // 新增
  phone: string;            // 新增
}

export interface OperationLogListParams {
  page: number;
  pageSize: number;
  keyword?: string;
}

export interface OperationLogListResponse {
  code: number;
  msg: string;
  total: number;
  data: OperationLog[];
}

export interface OperationLogDetailResponse {
  code: number;
  msg: string;
  data: OperationLogDetail;
}

// 获取操作日志列表
export async function getOperationLogList(params: OperationLogListParams) {
  return requestClient.get<OperationLogListResponse>('/system/log/operation', {
    params,
  });
}

// 获取操作日志详情
export async function getOperationLogDetail(id: number) {
  return requestClient.get<OperationLogDetailResponse>('/system/log/operation/detail', {
    params: { id },
  });
}
```

### 2. 列表页面修改

修改操作日志列表页面（假设是 `src/views/system/log/operation.vue`）：

```vue
<template>
  <div class="operation-log-page">
    <!-- 搜索栏 -->
    <div class="search-bar">
      <a-input-search
        v-model:value="searchKeyword"

    <!-- 表格 -->
    <a-table

      :data-source="logList"

      <!-- 操作地点列 -->
      <template #location="{ record }">
        <span>{{ record.location || '未知' }}</span>

        <a-descriptions-item label="操作人">

        <a-descriptions-item label="用户名">
          {{ detailData.username }}
        </a-descriptions-item>
        <a-descriptions-item label="邮箱">
          {{ detailData.email || '-' }}
        </a-descriptions-item>
        <a-descriptions-item label="手机号">
          {{ detailData.phone || '-' }}
        </a-descriptions-item>
        <a-descriptions-item label="操作时间">
          {{ detailData.created_at }}
        </a-descriptions-item>
        <a-descriptions-item label="操作模块">
          {{ detailData.module }}
        </a-descriptions-item>
        <a-descriptions-item label="操作类型">
          {{ detailData.action }}
        </a-descriptions-item>
        <a-descriptions-item label="请求方法">
          {{ detailData.method }}
        </a-descriptions-item>
        <a-descriptions-item label="IP地址">
          {{ detailData.ip }}
        </a-descriptions-item>
        <a-descriptions-item label="操作地点">
          {{ detailData.location || '未知' }}
        </a-descriptions-item>
        <a-descriptions-item label="操作状态">
          <a-tag :color="detailData.status === 1 ? 'success' : 'error'">
            {{ detailData.status === 1 ? '成功' : '失败' }}
          </a-tag>
        </a-descriptions-item>
        <a-descriptions-item label="执行时长">
          {{ detailData.duration }}ms
        </a-descriptions-item>
        <a-descriptions-item label="请求URL" :span="2">
          {{ detailData.url }}
        </a-descriptions-item>
        <a-descriptions-item label="请求参数" :span="2">
          <pre style="max-height: 200px; overflow: auto;">{{ detailData.params }}</pre>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue';
import { message } from 'ant-design-vue';
import { getOperationLogList, getOperationLogDetail, type OperationLog, type OperationLogDetail } from '@/api/system/log';

// 搜索关键词
const searchKeyword = ref('');

// 表格数据
const logList = ref<OperationLog[]>([]);
const loading = ref(false);

// 分页配置
const pagination = reactive({
  current: 1,
  pageSize: 20,
  total: 0,
  showSizeChanger: true,
  showTotal: (total: number) => `共 ${total} 条`,
});

// 表格列配置
const columns = [
  {
    title: '操作人',
    dataIndex: 'real_name',
    key: 'real_name',
    width: 120,
    slots: { customRender: 'real_name' },
  },
  {
    title: '操作时间',
    dataIndex: 'created_at',
    key: 'created_at',
    width: 180,
  },
  {
    title: '操作模块',
    dataIndex: 'module',
    key: 'module',
    width: 150,
  },
  {
    title: '操作类型',
    dataIndex: 'action',
    key: 'action',
    width: 100,
  },
  {
    title: 'IP地址',
    dataIndex: 'ip',
    key: 'ip',
    width: 150,
  },
  {
    title: '操作地点',
    dataIndex: 'location',
    key: 'location',
    width: 120,
    slots: { customRender: 'location' },
  },
  {
    title: '操作状态',
    dataIndex: 'status',
    key: 'status',
    width: 100,
    slots: { customRender: 'status' },
  },
  {
    title: '执行时长',
    dataIndex: 'duration',
    key: 'duration',
    width: 100,
    customRender: ({ text }: any) => `${text}ms`,
  },
  {
    title: '操作',
    key: 'action',
    width: 100,
    fixed: 'right',
    slots: { customRender: 'action' },
  },
];

// 详情弹窗
const detailVisible = ref(false);
const detailData = ref<OperationLogDetail | null>(null);

// 加载列表数据
const loadData = async () => {
  loading.value = true;
  try {
    const res = await getOperationLogList({
      page: pagination.current,
      pageSize: pagination.pageSize,
      keyword: searchKeyword.value,
    });
    
    if (res.code === 0) {
      logList.value = res.data;
      pagination.total = res.total;
    } else {
      message.error(res.msg || '获取日志列表失败');
    }
  } catch (error) {
    message.error('获取日志列表失败');
    console.error(error);
  } finally {
    loading.value = false;
  }
};

// 搜索
const handleSearch = () => {
  pagination.current = 1;
  loadData();
};

// 表格变化
const handleTableChange = (pag: any) => {
  pagination.current = pag.current;
  pagination.pageSize = pag.pageSize;
  loadData();
};

// 查看详情
const showDetail = async (id: number) => {
  try {
    const res = await getOperationLogDetail(id);
    if (res.code === 0) {
      detailData.value = res.data;
      detailVisible.value = true;
    } else {
      message.error(res.msg || '获取日志详情失败');
    }
  } catch (error) {
    message.error('获取日志详情失败');
    console.error(error);
  }
};

// 初始化
onMounted(() => {
  loadData();
});
</script>
```

## 三、关键修改点总结

### 1. 新增字段
- `real_name`：用户真实姓名（列表和详情）
- `location`：操作地点（列表和详情）
- `email`：用户邮箱（仅详情）
- `phone`：用户手机号（仅详情）

### 2. 接口调整
- 列表接口：添加 `keyword` 参数支持搜索
- 新增详情接口：`GET /api/system/log/operation/detail?id={id}`

### 3. 页面展示优化
- 操作人优先显示真实姓名，没有则显示用户名
- 新增操作地点列
- 详情弹窗展示更完整的信息（邮箱、手机号等）
